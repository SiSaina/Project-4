-- Create Database
IF NOT EXISTS (SELECT name FROM sys.databases WHERE name = N'BookShop')
BEGIN
    CREATE DATABASE BookShop;
END
GO

USE BookShop;
GO

-- Country Table
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Countries' AND xtype='U')
BEGIN
    CREATE TABLE Countries (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        Name NVARCHAR(100) NOT NULL UNIQUE
    );
END
GO

-- Theme Table
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Themes' AND xtype='U')
BEGIN
    CREATE TABLE Themes (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        Name NVARCHAR(100) NOT NULL UNIQUE
    );
END
GO

-- Shop Table
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Shops' AND xtype='U')
BEGIN
    CREATE TABLE Shops (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        Name NVARCHAR(200) NOT NULL,
        CountryId INT NOT NULL,
        FOREIGN KEY (CountryId) REFERENCES Countries(Id)
    );
END
GO

-- Author Table
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Authors' AND xtype='U')
BEGIN
    CREATE TABLE Authors (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        Name NVARCHAR(100) NOT NULL,
        Surname NVARCHAR(100) NOT NULL,
        CountryId INT NOT NULL,
        FOREIGN KEY (CountryId) REFERENCES Countries(Id)
    );
END
GO

-- Book Table
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Books' AND xtype='U')
BEGIN
    CREATE TABLE Books (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        Name NVARCHAR(200) NOT NULL,
        Pages INT NOT NULL,
        Price DECIMAL(10,2) NOT NULL,
        PublishDate DATE NOT NULL,
        AuthorId INT NOT NULL,
        ThemeId INT NOT NULL,
        FOREIGN KEY (AuthorId) REFERENCES Authors(Id),
        FOREIGN KEY (ThemeId) REFERENCES Themes(Id)
    );
END
GO

-- Sale Table
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Sales' AND xtype='U')
BEGIN
    CREATE TABLE Sales (
        Id INT IDENTITY(1,1) PRIMARY KEY,
        Price DECIMAL(10,2) NOT NULL,
        Quantity INT NOT NULL,
        SaleDate DATE NOT NULL,
        BookId INT NOT NULL,
        ShopId INT NOT NULL,
        FOREIGN KEY (BookId) REFERENCES Books(Id),
        FOREIGN KEY (ShopId) REFERENCES Shops(Id)
    );
END
GO
